## STEP 6 — API Layer (auth.api.js)

=> API Layer sirf backend se baat karegi
=> `axios.create()` se ek custom instance banate hain — baseURL aur withCredentials ek baar set karo!
=> Install karo: `npm i axios`


-------------------------------------

-------------------------------------

// =============== API LAYER ===============

import axios from "axios";
// axios import kiya — HTTP requests karne ke liye

const api = axios.create({
  baseURL: "http://localhost:3000/api/auth",
  // baseURL = har request ke aage yeh URL automatically lag jaayega
  // eg: api.post("/login") => "http://localhost:3000/api/auth/login"

  withCredentials: true,
  // cookies automatically bhejenge aur receive karenge
  // bina iske token cookie mein set nahi hoga!
});

// =========================================================
// FUNCTION 1: login
// Kaam: Backend pe POST /api/auth/login call karo
// =========================================================
export async function login(username, password) {
  const response = await api.post("/login", { username, password });
  return response.data;
  // response.data => backend ne jo JSON bheja woh return karo
  // eg: { message: "User LoggedIn Successfully", user: {...} }
}

// =========================================================
// FUNCTION 2: register
// Kaam: Backend pe POST /api/auth/register call karo
// =========================================================
export async function register(username, email, password) {
  const response = await api.post("/register", { username, email, password });
  return response.data;
  // eg: { message: "User Registered Successfully", user: {...} }
}

// =========================================================
// FUNCTION 3: getMe
// Kaam: Currently logged in user ka data fetch karo
// =========================================================
export async function getMe() {
  const response = await api.get("/get-me");
  // cookie mein token hai => backend identify karega ki kaun hai
  return response.data;
  // eg: { user: { username: "rahul", email: "...", ... } }
}

import { createContext, useState } from "react";
// createContext => context banane ke liye
// useState => data store karne ke liye

export const AuthContext = createContext();
// AuthContext banao — koi bhi component isko useContext se read kar sakta hai

export const AuthProvider = ({ children }) => {
// AuthProvider => wrapper component hai
// jo bhi iske andar ho woh sab context read kar sakte hain

  const [user, setUser] = useState(null);
  // user = currently logged in user ka data
  // null = abhi koi logged in nahi

    {/* value mein jo daalo woh poori app mein available hoga */}
    {/* user, setUser => user ka data aur use update karne ka function */}
    {/* loading, setLoading => loading state aur use update karne ka function */}
    {/* NOTE: setUser aur setLoading Hooks Layer use karega — UI nahi! */}

      {children}
      {/* jo bhi components AuthProvider ke andar honge */}
    </AuthContext.Provider>
  );
};

## STEP 8 — Hooks Layer (useAuth.js)

=> Hooks Layer = Manager
=> Yeh State aur API dono ko manage karta hai
=> UI seedha State ya API se baat nahi karega — Hook ke through karega!

=> Yahan handleLogin, handleRegister, handleGetMe likhenge
=> Yeh functions API call karenge aur response State mein save karenge


--------------------------------------------
src > features > auth > hooks > useAuth.js
--------------------------------------------

// =============== HOOKS LAYER ===============

import { useContext } from "react";
// useContext => context read karne ke liye

import { AuthContext } from "../auth.context";
// State Layer import kiya — user, setUser, loading, setLoading milega

import { login, register, getMe } from "../services/auth.api";
// API Layer import kiya — login, register, getMe functions milenge

export const useAuth = () => {

  const context = useContext(AuthContext);
  // context = { user, setUser, loading, setLoading }
  // State Layer ki saari values nikal lo

  const { user, setUser, loading, setLoading } = context;
  // destructure karo — individual variables mil jaayenge

  // =========================================================
  // handleLogin — login ka kaam
  // =========================================================
  const handleLogin = async (username, password) => {
    setLoading(true);
    // State Layer mein loading true karo — UI mein "Loading..." dikhega

    const response = await login(username, password);
    // API Layer ka login() call karo — backend pe POST /api/auth/login jaayega

    setUser(response.user);
    // response mein jo user data aaya — State Layer mein save karo
    // ab koi bhi component user read kar sakta hai

    setLoading(false);
    // API call khatam — loading false karo
  };

  // =========================================================
  // handleRegister — register ka kaam
  // =========================================================
  const handleRegister = async (username, email, password) => {
    setLoading(true);

    const response = await register(username, email, password);
    // API Layer ka register() call karo

    setUser(response.user);
    // registered user ko State mein save karo

    setLoading(false);
  };

  // =========================================================
  // handleGetMe — currently logged in user ka data fetch karo
  // =========================================================
  const handleGetMe = async () => {
    setLoading(true);

    const response = await getMe();
    // API Layer ka getMe() call karo — cookie se user identify hoga

    setUser(response.user);
    // fetched user State mein save karo

    setLoading(false);
  };

  return { user, setUser, loading, setLoading, handleLogin, handleRegister, handleGetMe };
  // yeh sab UI Layer ko de do
  // UI { handleLogin, loading, user } destructure karke use karega
};

import React, { useState } from "react";
import "../style/form.scss";
import { Link, useNavigate } from "react-router";
// useNavigate => programmatically navigate karne ke liye

import { useAuth } from "../hooks/useAuth";
// useAuth hook import kiya — Hooks Layer se
// handleLogin aur loading iske through milega

const Login = () => {

  // =========================================================
  // HOOKS
  // =========================================================
  const { user, loading, handleLogin } = useAuth();
  // handleLogin => Hooks Layer se mila — login ka kaam karega
  // loading => State Layer se mila — API chal rahi hai ki nahi

  const navigate = useNavigate();
  // navigate("/") => "/" pe jaao bina page reload ke

  // =========================================================
  // LOCAL STATE
  // =========================================================
  const [username, setUsername] = useState("");
  const [password, setPassword] = useState("");

  // =========================================================
  // handleSubmit — jab Login button dabaya jaaye
  // =========================================================
  const handleSubmit = async (e) => {
    e.preventDefault();
    // page reload rokko

    try {
      await handleLogin(username, password);
      // Hooks Layer ka handleLogin call karo
      // internally:
      //   1. setLoading(true) => State update
      //   2. API Layer ka login() call karo
      //   3. setUser(response.user) => State update
      //   4. setLoading(false) => State update

      console.log("user loggedIn");
      navigate("/");
      // login successful => home page pe bhejo

    } catch (err) {
      console.log(err);
      setUsername("");
      setPassword("");
      // error aaya => fields empty karo
    }
  };

  // =========================================================
  // LOADING CHECK
  // =========================================================
  if (loading) {
    return (
      <main>
        <h1>Loading...</h1>
        {/* API call chal rahi hai => Loading dikhao */}
      </main>
    );
  }

  return (
    <main>
      <div className="form-container">
        <h1>Login</h1>
        <form onSubmit={handleSubmit}>

          <input
            value={username}
            // value prop zaroori hai — React controlled input ke liye
            onInput={(e) => setUsername(e.target.value)}
            // user type kare => username state update karo
            type="text"
            name="username"
            id="username"
            placeholder="Enter Username"
          />

          <input
            value={password}
            onInput={(e) => setPassword(e.target.value)}
            type="password"
            name="password"
            id="password"
            placeholder="Enter password"
          />

          <button className="button primary-button">Login</button>

        </form>
        <p>
          Don't have an account? <Link to="/register">Register</Link>
        </p>
      </div>
    </main>
  );
};

export default Login;
