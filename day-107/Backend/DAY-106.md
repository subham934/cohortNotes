const mongoose = require("mongoose");

const followSchema = new mongoose.Schema(
  {
    follower: {
      type: String,   // username store hoga, ObjectId nahi
    },
    followee: {
      type: String,   // username store hoga, ObjectId nahi
    },
    status: {
      type: String,
      default: "pending",   // jab bhi follow request aayegi, default "pending" hoga
      enum: {
        values: ["pending", "accepted", "rejected"],
        message: "Status must be either 'pending' or 'accepted' or 'rejected'"

  },
  {
    timestamps: true,
  },
);

// unique index — ek user ek doosre ko sirf ek baar follow kar sake
// follower + followee ka combination unique hona chahiye
followSchema.index({ follower: 1, followee: 1 }, { unique: true });

const followModel = mongoose.model("follows", followSchema);

module.exports = followModel;

const followModel = require("../models/follow.model");
const userModel = require('../models/user.model');

// =========================================================
// FUNCTION 1: followUserController
// Kaam: Ek user ko follow karo
// Real life: Rahul ne Priya ko follow request bheja
// =========================================================
async function followUserController(req, res) {

  // LINE 1: Middleware se follower ka username nikalo
  // identifyUser middleware ne req.user set kiya tha
  // token mein ab username bhi hai isliye req.user.username se directly mil jaata hai
  // eg: Rahul logged in hai => followerUsername = "rahul"
  const followerUsername = req.user.username;

  // LINE 2: URL se followee ka username nikalo
  // req.params.username => URL mein jo ":username" tha woh yahan milta hai
  // eg: POST /api/users/follow/priya => followeeUsername = "priya"
  const followeeUsername = req.params.username;

  // LINE 3: Khud ko follow nahi kar sakte — check karo
  // agar dono same hain toh 400 Bad Request bhejo
  if (followerUsername === followeeUsername) {
    return res.status(400).json({
      message: "You cannot follow yourself!",
    });
  }

  // LINE 4: Check karo ki jisko follow karna hai woh user exist karta hai ya nahi
  // userModel.findOne({ username: followeeUsername }) => DB mein dhundo
  // agar nahi mila => 404 Not Found bhejo
  const isFolloweeUser = await userModel.findOne({ username: followeeUsername });

  if (!isFolloweeUser) {
    return res.status(404).json({
      message: `User ${followeeUsername} not found!`,
    });
  }

  // LINE 5: Check karo ki already follow kar chuka hai ya nahi
  // followModel.findOne({ follower, followee }) => DB mein dhundo
  // agar mila => 409 Conflict bhejo (already following)
  const isAlreadyFollowing = await followModel.findOne({
    follower: followerUsername,
    followee: followeeUsername,
  });

  if (isAlreadyFollowing) {
    return res.status(409).json({
      message: `You are already following ${followeeUsername}!`,
      follow: isAlreadyFollowing,
    });
  }

  // LINE 6: Sab check pass — follow record create karo DB mein
  // follower => jo follow kar raha hai (req.user.username)
  // followee => jisko follow kiya ja raha hai (req.params.username)
  // status   => automatically "pending" hoga — model mein default set hai
  const followRecord = await followModel.create({
    follower: followerUsername,
    followee: followeeUsername,
    // status: "pending" — yeh automatically set hoga, likhna zaruri nahi
  });

  // LINE 7: Success response bhejo
  // 201 = "Created" — naya resource successfully bana
  res.status(201).json({
    message: `You are now following ${followeeUsername}!`,
    follow: followRecord,
  });
}

// FUNCTION 2: unfollowUserController
// Kaam: Ek user ko unfollow karo
// Real life: Rahul ne Priya ko unfollow kiya
// =========================================================
async function unfollowUserController(req, res) {

  // LINE 1: Middleware se follower ka username nikalo
  // eg: Rahul unfollow button dabata hai => followerUsername = "rahul"
  const followerUsername = req.user.username;

  // LINE 2: URL se followee ka username nikalo
  // eg: POST /api/users/unfollow/priya => followeeUsername = "priya"
  const followeeUsername = req.params.username;

  // LINE 3: Check karo ki follow record exist karta hai ya nahi
  // agar nahi karta => matlab tu already unfollow kar chuka hai ya kabhi follow kiya hi nahi
  const isUserFollowing = await followModel.findOne({
    follower: followerUsername,
    followee: followeeUsername,
  });

  if (!isUserFollowing) {
    return res.status(404).json({
      message: `You are not following ${followeeUsername}!`,
    });
  }

  // LINE 4: Follow record delete karo DB se
  // findByIdAndDelete(isUserFollowing._id) => us specific document ko delete karo
  // isUserFollowing._id => woh document jo LINE 3 mein mila tha, uska id use karo
  await followModel.findByIdAndDelete(isUserFollowing._id);

  // LINE 5: Success response bhejo
  res.status(200).json({
    message: `You have unfollowed ${followeeUsername}!`,
  });
}

=> Ab Accept / Reject feature samajhte hain.

=> Ek important cheez pehle samajhlo:

  FOLLOW karte waqt:            ACCEPT/REJECT karte waqt:
  Rahul request karta hai       Priya request karti hai
  req.user = "rahul"            req.user = "priya"
  req.params = "priya"          req.params = "rahul"

=> Kyun ulta hai?
  - Follow mein RAHUL logged in tha aur usne button dabaya
  - Accept mein PRIYA logged in hai aur usne button dabaya
  - req.user hamesha wahi hota hai jo ABHI logged in hai!

=> Real life step by step:

  STEP 1 — Rahul apne phone se:
  POST /api/users/follow/priya
  req.user.username = "rahul"    (Rahul ka token)
  req.params.username = "priya"  (URL mein priya)
  DB mein save hua:
  { follower: "rahul", followee: "priya", status: "pending" }

  STEP 2 — Priya apne phone se:
  POST /api/users/accept/rahul
  req.user.username = "priya"    (Priya ka token)
  req.params.username = "rahul"  (URL mein rahul)
  DB mein update hua:
  { follower: "rahul", followee: "priya", status: "accepted" }


// =========================================================
// FUNCTION 3: acceptFollowController
// Kaam: Follow request accept karo
// Real life: Priya ne Rahul ki follow request accept ki
// =========================================================
async function acceptFollowController(req, res) {

  // LINE 1: Middleware se followee ka username nikalo
  // Priya logged in hai aur usne accept button dabaya
  // isliye req.user.username = "priya" — yahi followee hai jo accept kar rahi hai
  const followeeUsername = req.user.username;

  // LINE 2: URL se follower ka username nikalo
  // Priya ne URL mein Rahul ka naam diya:
  // POST /api/users/accept/rahul => followerUsername = "rahul"
  // yahi woh banda hai jisne pehle follow request bheja tha
  const followerUsername = req.params.username;

  // LINE 3: DB mein us follow record ko dhundo aur status update karo
  // findOneAndUpdate — 3 arguments hain:
  //
  //   1st argument — kaun sa document dhundna hai:
  //   { follower: "rahul", followee: "priya" }
  //   DB mein jaake woh specific document dhundo
  //
  //   2nd argument — kya update karna hai:
  //   { status: "accepted" }
  //   us document ka status "pending" se "accepted" kar do
  //
  //   3rd argument — options:
  //   { new: true } => updated document return karo, purana nahi
  //   by default MongoDB purana document return karta hai update se pehle wala
  //   new: true bolne se naya updated document milega
  const followRecord = await followModel.findOneAndUpdate(
    { follower: followerUsername, followee: followeeUsername },
    { status: "accepted" },
    { new: true }
  );

  // LINE 4: Agar follow record hi nahi mila toh 404 bhejo
  // matlab Rahul ne kabhi Priya ko follow kiya hi nahi tha
  if (!followRecord) {
    return res.status(404).json({
      message: "Follow request not found!",
    });
  }

  // LINE 5: Success response bhejo
  // followRecord mein ab status "accepted" hoga
  res.status(200).json({
    message: `You accepted ${followerUsername}'s follow request!`,
    follow: followRecord,
  });
}

// FUNCTION 4: rejectFollowController
// Kaam: Follow request reject karo
// Real life: Priya ne Rahul ki follow request reject ki

async function rejectFollowController(req, res) {

  // LINE 1: Middleware se followee ka username nikalo
  // Priya logged in hai aur usne reject button dabaya
  // isliye req.user.username = "priya"
  const followeeUsername = req.user.username;

  // LINE 2: URL se follower ka username nikalo
  // POST /api/users/reject/rahul => followerUsername = "rahul"
  const followerUsername = req.params.username;

  // LINE 3: DB mein us follow record ko dhundo aur status update karo
  // accept se bilkul same logic — sirf "accepted" ki jagah "rejected" hai
  const followRecord = await followModel.findOneAndUpdate(
    { follower: followerUsername, followee: followeeUsername },
    { status: "rejected" },
    { new: true }
  );

  // LINE 4: Agar follow record hi nahi mila toh 404 bhejo
  if (!followRecord) {
    return res.status(404).json({
      message: "Follow request not found!",
    });
  }

  // LINE 5: Success response bhejo
  res.status(200).json({
    message: `You rejected ${followerUsername}'s follow request!`,
    follow: followRecord,
  });
}

// =========================================================
// FUNCTION 5: likePostController
// Kaam: Ek post ko like karo
// =========================================================
async function likePostController(req, res) {

  // LINE 1: Middleware se username nikalo
  // like model mein user field String hai isliye username use karo, id nahi!
  // agar req.user.id use kiya toh error aayega:
  // "Cast to ObjectId failed for value 'test_1' at path 'user'"
  // kyunki like model mein user field String hai, ObjectId nahi
  const username = req.user.username;

  // LINE 2: URL se postId nikalo
  // req.params.postId => URL mein jo ":postId" tha woh yahan milta hai
  // eg: POST /api/posts/like/64abc123 => postId = "64abc123"
  const postId = req.params.postId;

  // LINE 3: Check karo ki post exist karta hai ya nahi
  // findById(postId) => DB mein woh specific post dhundo
  const post = await postModel.findById(postId);

  if (!post) {
    return res.status(404).json({
      message: "Post not found.",
    });
  }

  // LINE 4: Check karo ki already like kar chuka hai ya nahi
  // likeModel.findOne({ user, post }) => DB mein dhundo
  const isAlreadyLiked = await likeModel.findOne({
    user: username,
    post: postId,
  });

  if (isAlreadyLiked) {
    return res.status(409).json({
      message: "Post already liked.",
    });
  }

  // LINE 5: Like record create karo DB mein
  // user => username (String)
  // post => postId (ObjectId — MongoDB automatically convert karega)
  const like = await likeModel.create({
    user: username,
    post: postId,
  });

  // LINE 6: Success response bhejo
  res.status(201).json({
    message: "Post liked successfully.",
    like,
  });
}
