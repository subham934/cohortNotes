import React, { useState } from "react";
// React import kiya — JSX use karne ke liye zaroori
// useState import kiya — input fields ki values store karne ke liye

import "../style/form.scss";
// form ka CSS import kiya

import axios from "axios";
// axios import kiya — HTTP requests karne ke liye
// eg: POST /api/auth/register


const Register = () => {

  // =========================================================
  // STATE — input fields ki values store karne ke liye
  // =========================================================

  const [username, setUsername] = useState("");
  // username = abhi input mein jo value hai
  // setUsername = username ki value change karne ka function
  // useState("") = shuru mein khaali string

  const [email, setEmail] = useState("");
  // email field ki value

  const [password, setPassword] = useState("");
  // password field ki value


  // =========================================================
  // handleSubmit — jab user "Register" button dabaye
  // =========================================================

  async function handleSubmit(e) {

    e.preventDefault();
    // by default form submit hone pe page reload hota hai
    // e.preventDefault() se page reload ROKK dete hain

    await axios
      .post(
        "http://localhost:3000/api/auth/register",
        // axios se POST request bhejo backend pe
        // URL = http://localhost:3000/api/auth/register
        // yahi woh API hai jo humne backend mein banai thi!
        {
          username,   // state se username bhejo
          email,      // state se email bhejo
          password,   // state se password bhejo
        },
        {
          withCredentials: true,
          // withCredentials: true — bahut zaroori hai!
          // by default axios cookies set nahi karta frontend pe
          // yeh lagane se backend ka token cookie mein set ho jaata hai!
        }
      )
      .then((res) => {
        // agar register successful raha =>
        console.log(res.data);

        // fields empty karo
        setUsername("");
        setEmail("");
        setPassword("");
      })
      .catch((err) => {
        // agar koi error aaya — eg: user already exists (409) =>
        console.log(err.response.data.message);

        // error pe bhi fields empty karo
        setUsername("");
        setEmail("");
        setPassword("");
      });
  }


  return (
    <>
      <main>
        <div className="form-container">

          <h1>Register</h1>

          <form onSubmit={handleSubmit}>
          {/* jab form submit ho => handleSubmit function chalao */}

            <input
              value={username}
              // value={username} — bahut zaroori hai!
              // bina value prop ke React input field ko control nahi kar sakta
              // setUsername("") karne ke baad bhi field empty nahi hogi!
              onInput={(e) => setUsername(e.target.value)}
              // user jab bhi kuch type kare =>
              // e.target.value = jo type kiya woh value
              // setUsername se username state update karo
              type="text"
              name="username"
              placeholder="Enter Username"
            />

            <input
              value={email}
              onInput={(e) => setEmail(e.target.value)}
              type="email"
              name="email"
              placeholder="Enter Email"
            />

            <input
              value={password}
              onInput={(e) => setPassword(e.target.value)}
              type="password"
              name="password"
              placeholder="Enter Password"
            />

            <button type="submit">Register</button>
            {/* type="submit" => form submit hoga jab click karein */}

          </form>

          <p>
            Already have an account?{" "}
            <Link className="toggleAuthForm" to="/login">
              Login
            </Link>
            {/* Link => /login pe jaao bina page reload ke */}
          </p>

        </div>
      </main>
    </>
  );
};

export default Register;

const express = require("express");
const cookieParser = require('cookie-parser');
const cors = require("cors");
// cors import kiya — Cross Origin Resource Sharing allow karne ke liye

const app = express();
app.use(express.json());
app.use(cookieParser());
app.use(cors({
  credentials: true,
  // credentials: true => cookies allow karo cross-origin requests mein
  // bina iske withCredentials: true kaam nahi karega frontend pe!

  origin: 'http://localhost:5173'
  // origin => sirf is URL se aane wali requests allow karo
  // yahi hamara React app ka URL hai
}));

const authRouter = require('./routes/auth.routes');
const postRouter = require('./routes/post.routes');
const userRouter = require('./routes/user.routes');

app.use('/api/auth', authRouter);
app.use('/api/posts', postRouter);
app.use('/api/users', userRouter);

module.exports = app;

================================================

=> withCredentials aur credentials: true ka connection:

  Frontend (axios):          Backend (cors):
  withCredentials: true  +   credentials: true
          |                         |
          v                         v
  "Main cookies          "Main cookies
   bhejne ke liye         accept karne ke
   ready hun!"            liye ready hun!"

  Dono saath hone chahiye — ek bhi miss hua toh cookies set nahi hongi!

import React, { useState } from "react";
// useState import kiya — input fields ki values store karne ke liye

import "../style/form.scss";
// form ka CSS import kiya

import { Link } from "react-router";
// Link import kiya — page reload ke bina navigate karne ke liye

import axios from "axios";
// axios import kiya — backend se baat karne ke liye


const Login = () => {

  // =========================================================
  // STATE
  // =========================================================

  const [username, setUsername] = useState("");
  // login mein sirf username aur password chahiye — email nahi!

  const [password, setPassword] = useState("");


  // =========================================================
  // handleSubmit — jab user "Login" button dabaye
  // =========================================================

  async function handleSubmit(e) {

    e.preventDefault();
    // page reload rokko

    await axios
      .post(
        "http://localhost:3000/api/auth/login",
        // login API ka URL — register se alag hai!
        {
          username,   // state se username bhejo
          password,   // state se password bhejo
          // email nahi chahiye login mein
        },
        {
          withCredentials: true,
          // cookies set karne ke liye zaroori!
        }
      )
      .then((res) => {
        // login successful =>
        console.log(res.data);

        // fields empty karo
        setUsername("");
        setPassword("");
      })
      .catch((err) => {
        // error aaya — eg: invalid password (401), user not found (404)
        console.log(err.response.data.message);

        // error pe bhi fields empty karo
        setUsername("");
        setPassword("");
      });
  }


  return (
    <>
      <main>
        <div className="form-container">

          <h1>Login</h1>

          <form onSubmit={handleSubmit}>

            <input
              value={username}
              onInput={(e) => setUsername(e.target.value)}
              type="text"
              name="username"
              placeholder="Enter Username"
            />
            {/* login mein email field nahi hai — sirf username! */}

            <input
              value={password}
              onInput={(e) => setPassword(e.target.value)}
              type="password"
              name="password"
              placeholder="Enter Password"
            />

            <button type="submit">Login</button>

          </form>

          <p>
            Don't have an account?{" "}
            <Link className="toggleAuthForm" to="/register">
              Register
            </Link>
            {/* Register page pe jaao */}
          </p>

        </div>
      </main>
    </>
  );
};

export default Login;
