import React from "react";

const FormGroup = ({ label, placeholder, value, onChange }) => {
// Props:
// label       => input ka label text (eg: "Email", "Password")
// placeholder => input mein hint text
// value       => 2-way binding ke liye — state se aata hai
// onChange    => jab user type kare — state update karo

  return (
    <div className="form-group">
      <label htmlFor={label}>{label}</label>
      {/* htmlFor={label} => label click karne pe input focus ho jaata hai */}

      <input
        value={value}
        // value={value} => React controlled input
        // state empty hone pe field bhi empty hogi!

        onChange={onChange}
        // onChange => parent se aaya function — state update karega

        type="text"
        // NOTE: yeh "text" type hai — Password ke liye alag handling chahiye!
        // neeche mistake section mein explain kiya hai

        name={label}
        id={label}
        placeholder={placeholder}
        required
      />
    </div>
  );
};

// FUNCTION 1: register
// =========================================================
export async function register({ email, username, password }) {
// Object destructuring use kiya — clean aur readable!
// register({ email, username, password }) call kar sakte hain

  const response = await api.post("/api/auth/register", {
    email, password, username,
  });
  return response.data;
  // eg: { message: "User registered successfully", user: {...} }
}

// =========================================================
// FUNCTION 2: login
// =========================================================
export async function login({ email, username, password }) {
  const response = await api.post("/api/auth/login", {
    email, username, password,
  });
  return response.data;
}

// =========================================================
// FUNCTION 3: getMe
// =========================================================
export async function getMe() {
  const response = await api.get("/api/auth/get-me");
  // cookie mein token hai => backend identify karega ki kaun hai
  return response.data;
}

// =========================================================
// FUNCTION 4: logout
// =========================================================
export async function logout() {
  const response = await api.post("/api/auth/logout");
  // NOTE: teri notes mein api.get use kiya tha — GALAT hai! ❌
  // Logout ke liye POST use karo — GET nahi!
  // GET request URL mein visible hoti hai — security risk!
  return response.data;

import { createContext, useState } from "react";
// createContext => context banane ke liye
// useState => data store karne ke liye

export const AuthContext = createContext();

export const AuthProvider = ({ children }) => {

  const [user, setUser] = useState(null);
  // user = currently logged in user ka data
  // null = abhi koi logged in nahi

  const [loading, setLoading] = useState(true);
  // NOTE: loading = true se shuru kiya — false nahi!
  //
  // Kyun true se shuru kiya?
  // App load hone pe hum nahi jaante ki user logged in hai ya nahi
  // getMe() call hoga — backend se check hoga
  // Jab tak getMe() complete nahi hota — Protected.jsx ko wait karna chahiye!
  //
  // Agar false se shuru karo:
  //   App load => user = null => loading = false
  //   Protected.jsx => user nahi => /login redirect!
  //   getMe() chalti bhi nahi — user unnecessarily logout ho jaata! ❌
  //
  // true se shuru karo:
  //   App load => loading = true
  //   Protected.jsx => "Loading..." dikhao — wait karo!
  //   getMe() chali => user mila => loading = false => Home page! ✅

  return (
    <AuthContext.Provider value={{ user, setUser, loading, setLoading }}>
    {/* value mein jo daalo woh poori app mein available hoga */}
    {/* user, setUser       => user ka data aur update function */}
    {/* loading, setLoading => loading state aur update function */}

      {children}
    </AuthContext.Provider>
  );
};

## STEP 9 — Hooks Layer banao


------------------------------------------------------
src > features > auth > hooks > useAuth.js  (Hooks Layer)
------------------------------------------------------

import { login, register, getMe, logout } from "../services/auth.api";
// API Layer ke functions import kiye

import { useContext, useEffect } from "react";
// useContext => context read karne ke liye
// useEffect => side effects ke liye — page load pe getMe() call karna

import { AuthContext } from "../auth.context";
// State Layer import kiya

export function useAuth() {

  const context = useContext(AuthContext);
  // context = { user, setUser, loading, setLoading }

  const { user, setUser, loading, setLoading } = context;
  // destructure karo — individual variables mil jaayenge

  // =========================================================
  // handleRegister — register ka kaam
  // =========================================================
  async function handleRegister({ username, email, password }) {
    setLoading(true);
    // State Layer mein loading true karo

    const data = await register({ username, email, password });
    // API Layer ka register() call karo — backend pe request jaayegi

    setUser(data.user);
    // response mein jo user data aaya — State Layer mein save karo

    setLoading(false);
    // API call khatam — loading false karo
  }

  // =========================================================
  // handleLogin — login ka kaam
  // =========================================================
  async function handleLogin({ username, email, password }) {
    setLoading(true);
    const data = await login({ username, email, password });
    setUser(data.user);
    setLoading(false);
  }

  // =========================================================
  // handleGetMe — currently logged in user ka data fetch karo
  // =========================================================
  async function handleGetMe() {
    setLoading(true);
    const data = await getMe();
    // API Layer ka getMe() call karo — cookie se user identify hoga
    setUser(data.user);
    setLoading(false);
  }

  // =========================================================
  // handleLogout — logout ka kaam
  // =========================================================
  async function handleLogout() {
    setLoading(true);
    await logout();
    // API Layer ka logout() call karo — cookie delete hogi, Redis blacklist hoga
    setUser(null);
    // State Layer mein user null karo — ab user logged out hai!
    setLoading(false);
  }

  // =========================================================
  // useEffect — page load pe getMe() call karo!
  // =========================================================
  useEffect(() => {
    handleGetMe();
    // [] => sirf ek baar chalega — component mount hone pe
    // yeh solve karta hai page reload problem!
    //
    // Problem bina useEffect ke:
    //   User login karta hai => user state mein save hota hai
    //   User page reload karta hai => React state RESET ho jaata hai!
    //   user = null => Protected.jsx => /login redirect! ❌
    //
    // useEffect ke saath:
    //   Page reload => useEffect chalta hai
    //   handleGetMe() => getMe() API call => cookie se user check karta hai
    //   Cookie valid hai => user data wapas aata hai => State mein save! ✅
    //   Protected.jsx => user hai => Home page dikhao! ✅
  }, []);

  return {
    user,
    loading,
    handleRegister,
    handleLogin,
    handleGetMe,
    handleLogout,
    // yeh sab UI Layer ko de do
    // UI { handleLogin, loading } destructure karke use karega
  };
}

import React from "react";
// NOTE: teri notes mein "Children" bhi import tha — zaroorat nahi! ❌
// import React, { Children } from "react" => galat
// import React from "react" => sahi ✅

import { useAuth } from "../hooks/useAuth";
// useAuth hook import kiya — user aur loading milega

import { Navigate } from "react-router";
// Navigate => programmatically redirect karne ke liye
// <Navigate to="/login" /> => /login pe redirect kar do

const Protected = ({ children }) => {
// children => jo bhi Protected ke andar wrap kiya woh — eg: <h1>Home</h1>
// sirf tab render hoga jab user logged in ho!

  const { user, loading } = useAuth();
  // user    => logged in user ka data (null agar logged in nahi)
  // loading => getMe() chal rahi hai ki nahi

  // Case 1: Loading chal rahi hai — getMe() check kar rahi hai
  if (loading) {
    return <h1>Loading...</h1>;
    // "Loading..." dikhao — redirect mat karo abhi!
    // getMe() complete hone do — phir decide hoga
  }

  // Case 2: Loading khatam, user nahi mila — logged in nahi!
  if (!user) {
    return <Navigate to="/login" />;
    // /login pe redirect karo — home page allow mat karo!
  }

  // Case 3: Loading khatam, user mila — logged in hai!
  return children;
  // children render karo — eg: <h1>Home</h1>
  // user ko home page dikhao! ✅
};

export default Protected;

## Page Reload Problem — Kyun aur Kaise Fix Kiya:

```
PROBLEM (bina useEffect ke):
  Page reload
      |
      v
  React state RESET
  user = null, loading = false
      |
      v
  Protected.jsx => user nahi => /login redirect! ❌
  (getMe() kabhi chali hi nahi!)

SOLUTION (useEffect ke saath):
  Page reload
      |
      v
  React state RESET
  user = null, loading = TRUE
      |
      v
  Protected.jsx => loading = true => "Loading..." dikhao! wait karo!
      |
      v
  useEffect => handleGetMe() => cookie se user check karo
      |
      | user mila => setUser(data.user) => loading = false
      v
  Protected.jsx => user hai => Home page! ✅
```

> **loading = true se start karna** — yahi sabse important trick hai!
> Isse Protected.jsx ko time milta hai getMe() complete hone ka! 💪
